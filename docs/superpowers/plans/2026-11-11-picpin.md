# PicPin Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 原生 Mac 应用：拖入多张图片垂直/水平拼接，可指定目标比例，逐张控制子图策略（等比撑满+保留顶/中/底，或完整显示留白），导出或复制到剪贴板。

**Architecture:** 共享 `Composition` 纯数据模型 + `LayoutEngine` 纯函数布局（归一化几何），SwiftUI 预览与 Core Graphics 导出共用同一份布局结果（方案 A，双渲染器）。

**Tech Stack:** Swift 5.9+ / SwiftUI + AppKit / CoreGraphics / ImageIO，Swift Package 管理，无第三方依赖。

**Spec:** `docs/superpowers/specs/2026-11-11-picpin-design.md`

## Global Constraints

- macOS 14+；Swift Package（`swift build` / `swift test` / `swift run`）；零第三方依赖。
- 布局归一化基准 `base = 1000`；`gap`/`margin` 以同一基准为单位的相对值。
- weights 恒和为 1；单条最小占比 `minWeight = 0.05`；数量不匹配时按均分兜底。
- 默认值：垂直拼接、比例 3:4、策略 `(fill, .center)`、gap=0、margin=0、白色背景。
- 坐标约定：`LayoutEngine` 输出**左上原点（y 向下）**；`sourceRect` 为 CGImage 像素坐标（同为左上原点）。
- UI 文案全部中文。
- 导出宽度选项 1080/1242/1440/2048/原始最大宽度；JPEG 质量 0.92。
- 每个任务结束必须 commit。

## Review Focus

执行者按任务做完测试后，对照本清单确认以下输入类都有测试钉住（已分别落到对应任务）：

1. **极端比例**（如 20:1 水平画布 + 竖图）→ `sourceRect` 不得越界/为负 → Task 4 `testExtremeRatioClipsSourceRectToImageBounds`。
2. **拖边界越界**（share 拖出 `[minWeight, S−minWeight]`）→ clamp 生效且权重和恒为 1 → Task 2 `testSetBoundaryClampsToMinWeight`。
3. **weights 与 items 数量漂移**（追加/删除/移动后）→ 均分兜底 + 同步重排、保留相对比例 → Task 2 `testNormalizedWeightsFallsBackToEven`、Task 7 move/remove 测试。
4. **1 像素图 / 远小于条带的图** → fill 放大不除零、几何有效 → Task 4 `testTinyImageProducesValidPlacement`。
5. **gap+margin 大于内容区** → 条带高度 clamp ≥ 0、无 NaN → Task 3 `testOversizedGapMarginClampsToZero`。

（EXIF 方向为第 6 个高风险点，Task 5 已有像素级测试锁定，不重复列入。）

```swift
import XCTest

final class CompositionTests: XCTestCase {

            let img = TestSupport.makeImage(width: 100, height: 50, color: (1, 0, 0))

        var c = Composition()
        c.items = makeItems(3)

        c.weights = [0.5]                               // 长度不齐 → 均分
        XCTAssertEqual(c.normalizedWeights.count, 3)
        XCTAssertEqual(c.normalizedWeights.reduce(0, +), 1, accuracy: 1e-9)
    }

    func testNormalizedWeightsScalesAndClamps() {
        var c = Composition()
        c.items = makeItems(3)
        c.weights = [2, 1, 1]                           // 归一化 → [0.5, 0.25, 0.25]
        XCTAssertEqual(c.normalizedWeights, [0.5, 0.25, 0.25], accuracy: 1e-9)

        XCTAssertGreaterThan(w[1], 0.5)                 // 扣减量来自其他条
    }

        var c = Composition()
        c.items = makeItems(21)                          // 21 * 0.05 > 1 → 无法全部满足

        XCTAssertEqual(c.normalizedWeights, Array(repeating: 1.0/21, count: 21), accuracy: 1e-9)

        c.items = makeItems(3)
        c.weights = [0.2, 0.3, 0.5]
```

```swift
import CoreGraphics
import SwiftUI

enum StitchDirection: String, CaseIterable, Identifiable {
    case vertical, horizontal
    var id: String { rawValue }
    var label: String { self == .vertical ? "垂直拼接" : "水平拼接" }
}

enum FitMode: String, CaseIterable, Identifiable {
    case fill, fit
    var id: String { rawValue }
    var label: String { self == .fill ? "等比放大撑满" : "完整显示（留白）" }
}

enum CropAlignment: String, CaseIterable, Identifiable {
    case start, center, end
    var id: String { rawValue }
    func label(for direction: StitchDirection) -> String {
        switch (self, direction) {
        case (.start, .vertical): return "保留顶部"
        case (.center, .vertical): return "保留中间"
        case (.end, .vertical): return "保留底部"
        case (.start, .horizontal): return "保留左侧"
        case (.center, .horizontal): return "保留中间"
        case (.end, .horizontal): return "保留右侧"
        }
    }
}

struct SlicePolicy: Equatable {
    var mode: FitMode = .fill
    var alignment: CropAlignment = .center
}

struct AspectRatio: Identifiable, Hashable {
    let id: String
    let w: Double
    let h: Double
    var value: Double { w / h }

    static func custom(_ w: Double, _ h: Double) -> AspectRatio {
        func f(_ v: Double) -> String { v == v.rounded() ? String(Int(v)) : String(format: "%.2f", v) }
        return AspectRatio(id: "自定义 \(f(w)):\(f(h))", w: w, h: h)
    }
}

enum RatioPresets {
    static let r3x4 = AspectRatio(id: "3:4", w: 3, h: 4)
    static let r9x16 = AspectRatio(id: "9:16", w: 9, h: 16)
    static let r1x1 = AspectRatio(id: "1:1", w: 1, h: 1)
    static let r4x3 = AspectRatio(id: "4:3", w: 4, h: 3)
    static let r16x9 = AspectRatio(id: "16:9", w: 16, h: 9)
    static let r235x1 = AspectRatio(id: "2.35:1", w: 2.35, h: 1)
    static let all: [AspectRatio] = [r3x4, r9x16, r1x1, r4x3, r16x9, r235x1]
```

```swift
import CoreGraphics
import Foundation

struct Composition {
    var direction: StitchDirection = .vertical
    var aspectRatio: AspectRatio = RatioPresets.default
    var items: [ImageItem] = []
    var weights: [Double] = []                  // 与 items 等长且全部 > 0 时生效，否则均分
    var policies: [UUID: SlicePolicy] = [:]
    var gap: Double = 0                          // 相对值，单位 = LayoutEngine.base
    var margin: Double = 0
    var backgroundColor: CGColor = CGColor(red: 1, green: 1, blue: 1, alpha: 1)

    static let minWeight = 0.05
}

extension Composition {
    var count: Int { items.count }

    func policy(for id: UUID) -> SlicePolicy {
        policies[id] ?? SlicePolicy()
    }

    /// 归一化权重：数量不匹配/非法时均分；每条 ≥ minWeight；总和恒为 1
    var normalizedWeights: [Double] {
        let n = items.count
        guard n > 0 else { return [] }
        let even = Array(repeating: 1.0 / Double(n), count: n)
        guard weights.count == n, weights.allSatisfy({ $0 > 0 }) else { return even }
        guard Double(n) * Self.minWeight <= 1.0 else { return even }   // 无法全部满足下限
        let sum = weights.reduce(0, +)
        var w = weights.map { $0 / sum }
        for _ in 0..<n {                                              // 迭代分摊 clamp 扣减
            let under = w.filter { $0 < Self.minWeight }
            if under.isEmpty { break }
            let deficit = under.reduce(0.0) { $0 + (Self.minWeight - $1) }
            let overIdx = w.indices.filter { w[$0] >= Self.minWeight }
            let overSum = overIdx.reduce(0.0) { $0 + w[$1] }
            guard overSum > 0 else { break }
            for i in w.indices {
                if w[i] < Self.minWeight { w[i] = Self.minWeight }
                else { w[i] -= deficit * (w[i] / overSum) }
            }
        }
        let s = w.reduce(0, +)
        return w.map { $0 / s }
    }

    /// 拖动第 index 与 index+1 条之间边界：share ∈ [0,1] 表示第一条在本对总占比中的份额
    mutating func setBoundary(_ index: Int, share: Double) {
        let n = items.count
        guard n > 1, index >= 0, index < n - 1 else { return }
        var w = normalizedWeights
        let i = index, j = index + 1
        let pairSum = w[i] + w[j]                       // 本对总量不变，其他条不动
        let lo = Self.minWeight
        guard pairSum >= 2 * lo else { return }          // 理论不可达（pairSum ≥ 2/N ≥ 2*0.05 当 N ≤ 10）
        let wi = min(max(share * pairSum, lo), pairSum - lo)
        w[i] = wi
        w[j] = pairSum - wi
        weights = w
    }
}
```

注意 `setBoundary` 中当 N > 10 时 `pairSum` 可能 < 0.1，此时 guard 直接返回不修改（边界情况下 UI 允许的拖动范围由 view 层用 minWeight 提示，引擎保持总和恒定优先）。

```swift
final class LayoutEngineTests: XCTestCase {

    private func assertRect(_ a: CGRect, _ b: CGRect, accuracy: CGFloat = 0.5,
                            file: StaticString = #filePath, line: UInt = #line) {
        XCTAssertLessThan(abs(a.minX - b.minX), accuracy, file: file, line: line)
        XCTAssertLessThan(abs(a.minY - b.minY), accuracy, file: file, line: line)
        XCTAssertLessThan(abs(a.width - b.width), accuracy, file: file, line: line)
        XCTAssertLessThan(abs(a.height - b.height), accuracy, file: file, line: line)
    }

        var c = Composition()
        c.items = [makeItem(100, 200)]
        let r = LayoutEngine.layout(c)
        XCTAssertEqual(r.canvasSize.width, 1000, accuracy: 0.001)
        XCTAssertEqual(r.canvasSize.height, 1000 * 4.0 / 3.0, accuracy: 0.001)
```

```swift
import CoreGraphics

struct SlicePlacement: Equatable {
    let itemID: UUID
    let destination: CGRect   // 左上原点，归一化画布坐标（宽 = LayoutEngine.base）
    let sourceRect: CGRect?   // CGImage 像素坐标（左上原点）；nil = 整图无裁剪
    let fittedFrame: CGRect   // fill == destination；fit 为条带内实际图像 frame（四周留白）
}

struct LayoutResult: Equatable {
    let canvasSize: CGSize
    let placements: [SlicePlacement]
}

enum LayoutEngine {
    static let base: CGFloat = 1000

    static func layout(_ c: Composition) -> LayoutResult {
        let W = base
        let H = base / CGFloat(c.aspectRatio.value)
        let canvas = CGSize(width: W, height: H)
        let n = c.items.count
        guard n > 0 else { return LayoutResult(canvasSize: canvas, placements: []) }

        let m = CGFloat(c.margin)
        let gap = CGFloat(c.gap)
        let weights = c.normalizedWeights
        let contentOrigin = CGPoint(x: m, y: m)
        let contentW = max(0, W - 2 * m)
        let contentH = max(0, H - 2 * m)

        var placements: [SlicePlacement] = []
        if c.direction == .vertical {
            let stripTotal = max(0, contentH - CGFloat(n - 1) * gap)
            var y = contentOrigin.y
            for (i, item) in c.items.enumerated() {
                let strip = CGRect(x: contentOrigin.x, y: y, width: contentW,
                                   height: stripTotal * CGFloat(weights[i]))
                placements.append(placement(item: item, policy: c.policy(for: item.id),
                                            strip: strip, direction: .vertical))
                y += strip.height + gap
            }
        } else {
            let stripTotal = max(0, contentW - CGFloat(n - 1) * gap)
            var x = contentOrigin.x
            for (i, item) in c.items.enumerated() {
                let strip = CGRect(x: x, y: contentOrigin.y,
                                   width: stripTotal * CGFloat(weights[i]), height: contentH)
                placements.append(placement(item: item, policy: c.policy(for: item.id),
                                            strip: strip, direction: .horizontal))
                x += strip.width + gap
            }
        }
        return LayoutResult(canvasSize: canvas, placements: placements)
    }

    /// 单条带内几何。坐标系：左上原点。
    /// fill：scale = max(...)，比值大的轴恰好贴合、另一轴溢出：
    ///   - 主轴（对齐生效轴：垂直→y，水平→x）溢出 → 按 policy.alignment 裁剪（本任务实现）
    ///   - 副轴溢出 → 居中裁剪（Task 4 测试驱动补上）
    private static func placement(item: ImageItem, policy: SlicePolicy,
                                  strip: CGRect, direction: StitchDirection) -> SlicePlacement {
        let imgW = CGFloat(item.cgImage.width)
        let imgH = CGFloat(item.cgImage.height)
        guard imgW > 0, imgH > 0, strip.width > 0, strip.height > 0 else {
            return SlicePlacement(itemID: item.id, destination: strip, sourceRect: nil, fittedFrame: strip)
        }
        switch policy.mode {
        case .fill:
            let scale = max(strip.width / imgW, strip.height / imgH)
            let eps: CGFloat = 0.5                          // 归一化单位下的显示容差
            var srcX: CGFloat = 0, srcY: CGFloat = 0
            var srcW = imgW, srcH = imgH
            var cropped = false
            // 主轴：垂直→y（对齐策略），水平→x
            if direction == .vertical, imgH * scale > strip.height + eps {
                srcH = strip.height / scale
                srcY = axisOrigin(alignment: policy.alignment, imageLen: imgH, keepLen: srcH)
                cropped = true
            }
            if direction == .horizontal, imgW * scale > strip.width + eps {
                srcW = strip.width / scale
                srcX = axisOrigin(alignment: policy.alignment, imageLen: imgW, keepLen: srcW)
                cropped = true
            }
            let src = cropped ? CGRect(x: srcX, y: srcY, width: srcW, height: srcH) : nil
            return SlicePlacement(itemID: item.id, destination: strip, sourceRect: src, fittedFrame: strip)
        case .fit:
            let scale = min(strip.width / imgW, strip.height / imgH)
            let w = imgW * scale, h = imgH * scale
            let frame = CGRect(x: strip.midX - w / 2, y: strip.midY - h / 2, width: w, height: h)
            return SlicePlacement(itemID: item.id, destination: strip, sourceRect: nil, fittedFrame: frame)
        }
    }

    private static func axisOrigin(alignment: CropAlignment, imageLen: CGFloat, keepLen: CGFloat) -> CGFloat {
        switch alignment {
        case .start: return 0
        case .center: return (imageLen - keepLen) / 2
        case .end: return imageLen - keepLen
        }
    }
}
```

```swift
import CoreGraphics
import ImageIO
import Foundation

enum ImageLoader {
    static let thumbnailMaxPixel = 1200

    static func loadItem(from url: URL) -> ImageItem? {
        let srcOptions = [kCGImageSourceShouldCache: true] as CFDictionary
        guard let src = CGImageSourceCreateWithURL(url as CFURL, srcOptions) else { return nil }
        guard CGImageSourceGetCount(src) > 0 else { return nil }
        guard let raw = CGImageSourceCreateImageAtIndex(src, 0, srcOptions) else { return nil }
        let props = CGImageSourceCopyPropertiesAtIndex(src, 0, nil) as? [CFString: Any]
        let orientation = (props?[kCGImagePropertyOrientation] as? NSNumber)?.uint32Value ?? 1
        let upright = normalize(raw, exifOrientation: orientation)
        let thumb = makeThumbnail(src) ?? upright
        return ImageItem(id: UUID(), cgImage: upright, thumbnail: thumb,
                         fileName: url.lastPathComponent)
    }

    /// EXIF 规范化。原子变换（CG y 向上，已验证）：
    /// rotate90CW  = translate(0, w) + rotate(-π/2)，输出画布 (h, w)
    /// rotate90CCW = translate(h, 0) + rotate(+π/2)，输出画布 (h, w)
    /// flipH = translate(w, 0) + scale(-1, 1)；flipV = translate(0, h) + scale(1, -1)
    /// EXIF: 2=flipH 3=rot180 4=flipV 5=flipH→CCW 6=CW 7=flipH→CW 8=CCW
    static func normalize(_ image: CGImage, exifOrientation o: UInt32) -> CGImage {
        guard (2...8).contains(o) else { return image }
        let w = CGFloat(image.width), h = CGFloat(image.height)
        let swapped = (5...8).contains(o)
        let cw = swapped ? h : w, ch = swapped ? w : h
        guard let ctx = CGContext(data: nil, width: Int(cw), height: Int(ch),
                                  bitsPerComponent: 8, bytesPerRow: 0,
                                  space: CGColorSpaceCreateDeviceRGB(),
                                  bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue) else { return image }
        switch o {
        case 2:
            ctx.translateBy(x: w, y: 0); ctx.scaleBy(x: -1, y: 1)
        case 3:
            ctx.translateBy(x: w, y: h); ctx.rotate(by: .pi)
        case 4:
            ctx.translateBy(x: 0, y: h); ctx.scaleBy(x: 1, y: -1)
        case 5:
            ctx.translateBy(x: h, y: 0); ctx.rotate(by: .pi / 2)
            ctx.translateBy(x: w, y: 0); ctx.scaleBy(x: -1, y: 1)
        case 6:
            ctx.translateBy(x: 0, y: w); ctx.rotate(by: -.pi / 2)
        case 7:
            ctx.translateBy(x: 0, y: w); ctx.rotate(by: -.pi / 2)
            ctx.translateBy(x: w, y: 0); ctx.scaleBy(x: -1, y: 1)
        case 8:
            ctx.translateBy(x: h, y: 0); ctx.rotate(by: .pi / 2)
        default:
            break
        }
        ctx.interpolationQuality = .high
        ctx.draw(image, in: CGRect(x: 0, y: 0, width: w, height: h))
        return ctx.makeImage() ?? image
    }

    static func makeThumbnail(_ src: CGImageSource) -> CGImage? {
        let opts: [CFString: Any] = [
            kCGImageSourceCreateThumbnailFromImageAlways: true,
            kCGImageSourceCreateThumbnailWithTransform: true,     // 自动应用 EXIF 方向
            kCGImageSourceThumbnailMaxPixelSize: thumbnailMaxPixel,
        ]
        return CGImageSourceCreateThumbnailAtIndex(src, 0, opts as CFDictionary)
    }
}
```

```swift
import CoreGraphics

enum CompositionRenderer {
    /// 按输出像素宽度渲染（高度 = 宽度 / 比例）。使用原始 CGImage，保证导出保真。
    static func render(_ c: Composition, outputWidth: Int) -> CGImage? {
        let layout = LayoutEngine.layout(c)
        let scale = CGFloat(outputWidth) / LayoutEngine.base
        let W = Int((layout.canvasSize.width * scale).rounded())
        let H = Int((layout.canvasSize.height * scale).rounded())
        guard W > 0, H > 0 else { return nil }
        guard let ctx = CGContext(data: nil, width: W, height: H,
                                  bitsPerComponent: 8, bytesPerRow: 0,
                                  space: CGColorSpaceCreateDeviceRGB(),
                                  bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue) else { return nil }
        ctx.setFillColor(c.backgroundColor)
        ctx.fill(CGRect(x: 0, y: 0, width: W, height: H))
        ctx.interpolationQuality = .high
        for p in layout.placements {
            guard let item = c.items.first(where: { $0.id == p.itemID }) else { continue }
            let img = p.sourceRect.flatMap { item.cgImage.cropping(to: $0) } ?? item.cgImage
            // 布局左上原点 → CG 左下原点
            let rect = CGRect(x: p.fittedFrame.minX * scale,
                              y: CGFloat(H) - (p.fittedFrame.minY + p.fittedFrame.height) * scale,
                              width: p.fittedFrame.width * scale,
                              height: p.fittedFrame.height * scale)
            ctx.draw(img, in: rect)
        }
        return ctx.makeImage()
    }
}
```

```swift
import AppKit
import CoreGraphics
import ImageIO
import UniformTypeIdentifiers

enum ImageExporter {
    enum Format: String, CaseIterable, Identifiable {
        case png = "PNG", jpeg = "JPEG"
        var id: String { rawValue }
    }

    static func encode(_ image: CGImage, format: Format) -> Data? {
        let ut: UTType = format == .png ? .png : .jpeg
        let out = NSMutableData()
        guard let dest = CGImageDestinationCreateWithData(out, ut.identifier as CFString, 1, nil) else { return nil }
        let props: [CFString: Any] = format == .jpeg
            ? [kCGImageDestinationLossyCompressionQuality: 0.92]
            : [:]
        CGImageDestinationAddImage(dest, image, props as CFDictionary)
        guard CGImageDestinationFinalize(dest) else { return nil }
        return out as Data
    }

    /// 同时写 PNG 与 TIFF，最大化粘贴兼容性（微信/小红书/文档）
    static func copyToClipboard(_ image: CGImage) {
        let pb = NSPasteboard.general
        pb.clearContents()
        var payload: [(Data, NSPasteboard.PasteboardType)] = []
        if let png = encode(image, format: .png) {
            payload.append((png, .png))
        }
        if let tiff = NSBitmapImageRep(cgImage: image).tiffRepresentation {
            payload.append((tiff, .tiff))
        }
        pb.setData(payload.first?.0, forType: payload.first?.1 ?? .tiff)
        for (d, t) in payload.dropFirst() { pb.setData(d, forType: t) }
    }
}
```

```swift
import AppKit
import Combine
import Foundation
import UniformTypeIdentifiers

@MainActor
final class CompositionStore: ObservableObject {
    @Published var composition = Composition()
    @Published var selectedID: UUID?
    @Published var exportWidth: Int = 1242                   // 0 = 原始最大宽度
    @Published var exportFormat: ImageExporter.Format = .png
    @Published var lastIgnoredCount = 0

    /// 追加图片：保留旧条相对比例，新条各占 1/(n+k)；非图片计数到 lastIgnoredCount
    func addImages(from urls: [URL]) {
        var loaded: [ImageItem] = []
        var ignored = 0
        for url in urls {
            if isImageURL(url), let item = ImageLoader.loadItem(from: url) {
                loaded.append(item)
            } else {
                ignored += 1
            }
        }
        lastIgnoredCount = ignored
        guard !loaded.isEmpty else { return }
        var c = composition
        let n = c.items.count, k = loaded.count, total = n + k
        var w = c.normalizedWeights.map { $0 * Double(n) / Double(total) }
        w += Array(repeating: 1.0 / Double(total), count: k)
        c.items += loaded
        c.weights = w
        composition = c
    }

    func move(from source: IndexSet, to destination: Int) {
        var items = composition.items
        var weights = composition.normalizedWeights
        items.move(fromOffsets: source, toOffset: destination)
        weights.move(fromOffsets: source, toOffset: destination)
        composition.items = items
        composition.weights = weights
    }

    func remove(at offsets: IndexSet) {
        var items = composition.items
        var weights = composition.normalizedWeights
        items.remove(atOffsets: offsets)
        weights.remove(atOffsets: offsets)
        composition.items = items
        composition.weights = weights
        let ids = Set(items.map(\.id))
        composition.policies = composition.policies.filter { ids.contains($0.key) }
        if let sel = selectedID, !ids.contains(sel) { selectedID = nil }
    }

    func resetToEven() {
        composition.weights = []                              // 均分兜底
    }

    func setBoundary(_ index: Int, share: Double) {
        composition.setBoundary(index, share: share)
    }

    var resolvedOutputWidth: Int {
        if exportWidth > 0 { return exportWidth }
        return composition.items.map { Int($0.pixelSize.width) }.max() ?? 1000
    }

    private func isImageURL(_ url: URL) -> Bool {
        guard let t = UTType(filenameExtension: url.pathExtension) else { return false }
        return t.conforms(to: .image)
    }
}
```

```swift
                    .frame(width: p.fittedFrame.width * scale,
                           height: p.fittedFrame.height * scale)
                    .position(x: p.fittedFrame.midX * scale,
                              y: p.fittedFrame.midY * scale)
            }
        }
        .frame(width: p.destination.width * scale,
               height: p.destination.height * scale)
        .position(x: p.destination.midX * scale, y: p.destination.midY * scale)
        .contentShape(Rectangle())
        .onTapGesture { store.selectedID = p.itemID }
        .overlay(
            RoundedRectangle(cornerRadius: 2)
                .strokeBorder(selected ? Color.accentColor : .clear, lineWidth: 3)
        )
    }

    @State private var dragBase: (pairSum: Double, share: Double)? = nil

    @ViewBuilder
    private func boundaryHandles(layout: LayoutResult, scale: CGFloat) -> some View {
        let n = store.composition.items.count
        let vertical = store.composition.direction == .vertical
        ForEach(0..<max(0, n - 1), id: \.self) { i in
            let prev = layout.placements[i].destination
            let x = vertical ? prev.midX : prev.maxX + CGFloat(store.composition.gap) / 2
            let y = vertical ? prev.maxY + CGFloat(store.composition.gap) / 2 : prev.midY
            BoundaryHandle(vertical: vertical)
                .position(x: x * scale, y: y * scale)
                .gesture(
                    DragGesture(minimumDistance: 2)
                        .onChanged { g in
                            let w = store.composition.normalizedWeights
                            let S: Double
                            let baseShare: Double
                            if let d = dragBase {
                                (S, baseShare) = (d.pairSum, d.share)
                            } else {
                                S = w[i] + w[i + 1]
                                baseShare = S > 0 ? w[i] / S : 0.5
                                dragBase = (S, baseShare)
                            }
                            // 内容区归一化长度（同 LayoutEngine 口径）
                            let nD = Double(n)
                            let stripTotalD = (vertical
                                ? Double(layout.canvasSize.height) - 2 * store.composition.margin
                                : Double(layout.canvasSize.width) - 2 * store.composition.margin)
                                - (nD - 1) * store.composition.gap
                            guard stripTotalD > 0, S > 0 else { return }
                            let deltaD = (vertical ? Double(g.translation.height)
                                                   : Double(g.translation.width)) / Double(scale) / stripTotalD
                            store.setBoundary(i, share: baseShare + deltaD / S)
                        }
                        .onEnded { _ in dragBase = nil }
                )
        }
    }
}

private struct BoundaryHandle: View {
    let vertical: Bool
    var body: some View {
        ZStack {
            // 抓取区
            Rectangle().fill(.clear)
                .frame(width: vertical ? 60 : 22, height: vertical ? 22 : 60)
                .contentShape(Rectangle())
            // 视觉线
            Capsule()
                .fill(Color.accentColor.opacity(0.85))
                .frame(width: vertical ? 44 : 3, height: vertical ? 3 : 44)
        }
        .onHover { inside in
            if inside { NSCursor.resizeUpDown.set() } else { NSCursor.arrow.set() }
        }
    }
}
```

```swift
            } label: {
```

```swift
            Button {

            } label: {
                Label("导出…", systemSymbol: "square.and.arrow.down")

            .buttonStyle(.borderedProminent)
            .disabled(store.composition.items.isEmpty)
        }
        .controlSize(.regular)

    private func renderedImage() -> CGImage? {
        CompositionRenderer.render(store.composition, outputWidth: store.resolvedOutputWidth)
    }

    private func copyToClipboard() {
        guard let img = renderedImage() else { return }
        ImageExporter.copyToClipboard(img)
        copied = true
        DispatchQueue.main.asyncAfter(deadline: .now() + 1.5) { copied = false }
    }

    private func exportToFile() {
        guard let img = renderedImage(),
              let data = ImageExporter.encode(img, format: store.exportFormat) else { return }
        let panel = NSSavePanel()
        panel.allowedContentTypes = [store.exportFormat == .png ? .png : .jpeg]
        let ext = store.exportFormat == .png ? "png" : "jpg"
        panel.nameFieldStringValue = "PicPin-\(Int(Date().timeIntervalSince1970)).\(ext)"
        guard panel.runModal() == .OK, let url = panel.url else { return }
        do {
            try data.write(to: url)
        } catch {
            let alert = NSAlert(error: error)
            alert.runModal()
        }
    }
}
```

````markdown
# PicPin

把多张图片拖进来，拼成一张图，并可指定目标比例。切比例时不粗暴裁剪——每张子图可以单独选择「等比放大撑满 + 保留顶/中/底」或「完整显示留白」。

6. 底部导出 PNG/JPEG（宽度可选）或直接复制到剪贴板
````
